local function makeDraggable(frame, handle)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function speak(text)
    if not text or text == "" then return end
    local clean = text:sub(1, 280)
    pcall(function()
        local tts = Instance.new("AudioTextToSpeech")
        tts.Parent = workspace
        tts.Text = clean
        tts.VoiceId = "3"
        tts.Volume = 1.2
        tts.Speed = 1.05

        local out = Instance.new("AudioDeviceOutput")
        out.Parent = workspace

        local wire = Instance.new("Wire")
        wire.SourceInstance = tts
        wire.TargetInstance = out
        wire.Parent = workspace

        tts:LoadAsync()
        tts:Play()

        task.delay(15, function()
            pcall(function()
                tts:Destroy()
                out:Destroy()
                wire:Destroy()
            end)
        end)
    end)
end

local history = {
    {role = "system", content = "You are Verity, a calm helpful companion inside Roblox. Keep answers short and natural, under 2 sentences when possible."}
}

local function askAI(userText)
    table.insert(history, {role = "user", content = userText})

    local body = HttpService:JSONEncode({
        model = "openai/gpt-oss-20b",
        messages = history,
        max_tokens = 150,
        temperature = 0.7
    })

    local reqFunc = nil
    if syn and syn.request then
        reqFunc = syn.request
    elseif http and http.request then
        reqFunc = http.request
    elseif request then
        reqFunc = request
    elseif http_request then
        reqFunc = http_request
    elseif fluxus and fluxus.request then
        reqFunc = fluxus.request
    end

    if not reqFunc then
        return "no request function on this executor"
    end

    local ok, res = pcall(function()
        return reqFunc({
            Url = "https://api.groq.com/openai/v1/chat/completions",
            Method = "POST",
            Headers = {
                ["Authorization"] = "Bearer " .. API_KEY,
                ["Content-Type"] = "application/json"
            },
            Body = body
        })
    end)

    if not ok then
        return "request error: " .. tostring(res)
    end

    if not res then
        return "empty response object"
    end

    local bodyStr = res.Body or res.body or res.ResponseBody or ""
    if bodyStr == "" then
        local status = res.StatusCode or res.status or res.Status or "?"
        return "empty body (status " .. tostring(status) .. ")"
    end

    local decodeOk, data = pcall(function()
        return HttpService:JSONDecode(bodyStr)
    end)

    if not decodeOk then
        return "bad json: " .. tostring(bodyStr):sub(1, 100)
    end

    if data.error then
        local msg = data.error.message or data.error or "unknown"
        return "api: " .. tostring(msg)
    end

    if data.choices and data.choices[1] and data.choices[1].message then
        local reply = data.choices[1].message.content
        table.insert(history, {role = "assistant", content = reply})
        if #history > 12 then
            table.remove(history, 2)
            table.remove(history, 2)
        end
        return reply
    end

    return "no reply in response"
end

local busy = false
local function send()
    if busy then return end
    local text = inputBox.Text:gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return end

    busy = true
    inputBox.Text = ""
    addMessage("You", text, YOU_COL)
    addMessage("Verity", "...", VERITY_COL)

    task.spawn(function()
        local reply = askAI(text)
        for _, child in ipairs(chatScroll:GetChildren()) do
            if child:IsA("Frame") and child.LayoutOrder == msgCount then
                child:Destroy()
                break
            end
        end
        msgCount -= 1
        addMessage("Verity", reply, VERITY_COL)
        speak(reply)
        busy = false
    end)
end
